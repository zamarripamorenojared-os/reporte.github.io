<script>
  let prestamos = JSON.parse(localStorage.getItem('fim_prestamos') || '[]');
  let tabActual = 'todos';
  let idEntregando = null;

  // Set defaults for date/time
  function setDefaults() {
    const now = new Date();
    const fechaStr = now.toISOString().split('T')[0];
    const horaStr = now.toTimeString().slice(0,5);
    document.getElementById('f-fecha-prestamo').value = fechaStr;
    document.getElementById('f-hora-prestamo').value = horaStr;
  }

  function guardar() {
    localStorage.setItem('fim_prestamos', JSON.stringify(prestamos));
  }

  function getEstado(p) {
    if (p.fechaEntrega) return 'entregado';
    const hoy = new Date();
    const fp = new Date(p.fechaPrestamo);
    const diff = (hoy - fp) / (1000*60*60*24);
    if (diff > 1 && !p.fechaEntrega) return 'vencido';
    return 'prestamo';
  }

  function registrarPrestamo() {
    const nombre = document.getElementById('f-nombre').value.trim();
    const equipo = document.getElementById('f-equipo').value.trim();
    if (!nombre || !equipo) { alert('Nombre y equipo son obligatorios.'); return; }

    const tipo = document.querySelector('input[name=tipo]:checked').value;
    const p = {
      id: Date.now(),
      nombre,
      carrera: document.getElementById('f-carrera').value,
      grupo: document.getElementById('f-grupo').value.trim(),
      tipo,
      cg: document.getElementById('f-cg').value,
      equipo,
      serie: document.getElementById('f-serie').value.trim(),
      cantidad: document.getElementById('f-cantidad').value || 1,
      fechaPrestamo: document.getElementById('f-fecha-prestamo').value,
      horaPrestamo: document.getElementById('f-hora-prestamo').value,
      obs: document.getElementById('f-obs').value.trim(),
      firmaEncargado: document.getElementById('f-firma-encargado').value.trim(),
      fechaEntrega: null, horaEntrega: null,
      firmaEntrega: null, firmaEncargado2: null, obsEntrega: null
    };
    prestamos.unshift(p);
    guardar();
    renderTabla();
    // Clear form
    ['f-nombre','f-grupo','f-equipo','f-serie','f-obs','f-firma-encargado'].forEach(id => document.getElementById(id).value = '');
    document.getElementById('f-cantidad').value = 1;
    document.getElementById('f-carrera').value = '';
    document.querySelector('input[name=tipo][value="A"]').checked = true;
    setDefaults();
  }

  function abrirEntrega(id) {
    idEntregando = id;
    const now = new Date();
    document.getElementById('m-fecha-entrega').value = now.toISOString().split('T')[0];
    document.getElementById('m-hora-entrega').value = now.toTimeString().slice(0,5);
    document.getElementById('m-firma-entrega').value = '';
    document.getElementById('m-firma-encargado2').value = '';
    document.getElementById('m-obs-entrega').value = '';
    document.getElementById('modal-entrega').classList.add('open');
  }

  function cerrarModal() {
    document.getElementById('modal-entrega').classList.remove('open');
    idEntregando = null;
  }

  function confirmarEntrega() {
    const p = prestamos.find(x => x.id === idEntregando);
    if (!p) return;
    p.fechaEntrega = document.getElementById('m-fecha-entrega').value;
    p.horaEntrega = document.getElementById('m-hora-entrega').value;
    p.firmaEntrega = document.getElementById('m-firma-entrega').value.trim();
    p.firmaEncargado2 = document.getElementById('m-firma-encargado2').value.trim();
    p.obsEntrega = document.getElementById('m-obs-entrega').value.trim();
    guardar();
    cerrarModal();
    renderTabla();
  }

  function eliminar(id) {
    if (!confirm('¿Eliminar este registro?')) return;
    prestamos = prestamos.filter(x => x.id !== id);
    guardar();
    renderTabla();
  }

  function setTab(tab, el) {
    tabActual = tab;
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    el.classList.add('active');
    renderTabla();
  }

  function fmtFecha(d, h) {
    if (!d) return '—';
    const [y,m,day] = d.split('-');
    return `${day}/${m}/${y}${h ? ' '+h : ''}`;
  }

  function badgeHTML(estado) {
    const map = {
      prestamo: ['badge-prestamo', '● En Préstamo'],
      entregado: ['badge-entregado', '✓ Entregado'],
      vencido: ['badge-vencido', '⚠ Vencido'],
    };
    const [cls, txt] = map[estado];
    return `<span class="badge ${cls}">${txt}</span>`;
  }

  function renderTabla() {
    const q = document.getElementById('search').value.toLowerCase();
    let data = prestamos.filter(p => {
      const estado = getEstado(p);
      if (tabActual !== 'todos' && estado !== tabActual) return false;
      if (q && ![p.nombre, p.equipo, p.serie, p.grupo, p.carrera].join(' ').toLowerCase().includes(q)) return false;
      return true;
    });

    // Stats
    const activos = prestamos.filter(p => getEstado(p) === 'prestamo').length;
    const entregados = prestamos.filter(p => getEstado(p) === 'entregado').length;
    document.getElementById('stat-active').textContent = activos;
    document.getElementById('stat-returned').textContent = entregados;
    document.getElementById('stat-total').textContent = prestamos.length;

    const tbody = document.getElementById('tabla-body');
    const empty = document.getElementById('empty-state');

    if (data.length === 0) {
      tbody.innerHTML = '';
      empty.style.display = 'block';
      return;
    }
    empty.style.display = 'none';

    tbody.innerHTML = data.map((p, i) => {
      const estado = getEstado(p);
      const tipoLabel = p.tipo === 'A' ? 'Alumno' : 'Profesor';
      return `<tr>
        <td class="equip-id">${String(i+1).padStart(3,'0')}</td>
        <td>
          <div style="font-weight:600">${p.nombre}</div>
          <div class="user-type">${tipoLabel} · ${p.cg}</div>
        </td>
        <td>${p.carrera || '—'} · ${p.grupo || '—'}</td>
        <td class="equip-name">${p.equipo}</td>
        <td class="equip-id">${p.serie || '—'}</td>
        <td style="text-align:center">${p.cantidad}</td>
        <td class="date-cell">${fmtFecha(p.fechaPrestamo, p.horaPrestamo)}</td>
        <td class="date-cell">${fmtFecha(p.fechaEntrega, p.horaEntrega)}</td>
        <td>${badgeHTML(estado)}</td>
        <td style="font-size:.75rem;color:var(--muted)">${p.firmaEncargado || '—'}</td>
        <td>
          <div class="actions">
            ${estado !== 'entregado' ? `<button class="btn btn-success btn-sm" onclick="abrirEntrega(${p.id})" title="Registrar entrega">✓</button>` : ''}
            <button class="btn btn-danger btn-sm" onclick="eliminar(${p.id})" title="Eliminar">✕</button>
          </div>
        </td>
      </tr>`;
    }).join('');
  }

  // Click outside modal to close
  document.getElementById('modal-entrega').addEventListener('click', function(e) {
    if (e.target === this) cerrarModal();
  });

  setDefaults();
  renderTabla();
</script>
